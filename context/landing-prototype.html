useDebounce hook

Waits until a value stops changing.

typescriptCopy
export function useDebounce(value, ms) {
  const [v, setV] = useState(value);
  useEffect(() => {
    const id = setTimeout(
      () => setV(value), ms);
    return () => clearTimeout(id);
  }, [value, ms]);
  return v;
}
